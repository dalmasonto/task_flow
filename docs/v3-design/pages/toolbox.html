#!/usr/bin/env bash
set -euo pipefail
gh workflow run deploy-backend.yml --ref "$branch"
sleep 20
run=$(gh run list -w deploy-backend.yml -L 1 --json databaseId -q '.[0].databaseId')
gh run watch "$run" --exit-status
curl -fsS "https://api.taskflow.supercodehive.com${verify_path:-/health}" > /dev/null
